cmake_minimum_required(VERSION 3.18)
project(clicalc VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CLICALC_BUILD_CLI "Build the terminal frontend" ON)
option(CLICALC_READLINE "Enable GNU Readline editing and history" ON)
include(CTest)
find_path(MPFR_INCLUDE_DIR mpfr.h REQUIRED)
find_path(GMP_INCLUDE_DIR gmpxx.h REQUIRED)
find_library(MPFR_LIBRARY mpfr REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
add_library(clicalc_core src/number.cpp src/parser.cpp src/functions.cpp src/units.cpp src/engine.cpp src/session.cpp)
target_include_directories(clicalc_core PUBLIC include ${MPFR_INCLUDE_DIR} ${GMP_INCLUDE_DIR})
target_link_libraries(clicalc_core PUBLIC ${MPFR_LIBRARY} ${GMPXX_LIBRARY} ${GMP_LIBRARY})
if(CLICALC_BUILD_CLI)
  add_executable(clicalc src/main.cpp src/terminal.cpp)
  target_link_libraries(clicalc PRIVATE clicalc_core)
  if(CLICALC_READLINE)
    find_path(READLINE_INCLUDE_DIR readline/readline.h REQUIRED)
    find_library(READLINE_LIBRARY readline REQUIRED)
    target_include_directories(clicalc PRIVATE ${READLINE_INCLUDE_DIR})
    target_link_libraries(clicalc PRIVATE ${READLINE_LIBRARY})
    target_compile_definitions(clicalc PRIVATE CLICALC_READLINE)
  endif()
  install(TARGETS clicalc RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE clicalc_core)
  add_test(NAME engine COMMAND engine_tests)
  add_executable(features_tests tests/features_tests.cpp)
  target_link_libraries(features_tests PRIVATE clicalc_core)
  add_test(NAME features COMMAND features_tests)
  if(CLICALC_BUILD_CLI)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
      add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py $<TARGET_FILE:clicalc>)
      if(CLICALC_READLINE AND UNIX)
        add_test(NAME terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_tests.py $<TARGET_FILE:clicalc>)
      endif()
    endif()
  endif()
endif()
